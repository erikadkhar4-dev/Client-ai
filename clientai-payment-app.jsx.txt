// Simulated QR grid (visual only)
function QRGrid({ size = 140 }) {
  const cells = 21;
  const [grid] = useState(() => {
    const g = [];
    for (let r = 0; r < cells; r++) {
      const row = [];
      for (let c = 0; c < cells; c++) {
        const isCorner = (r < 7 && c < 7) || (r < 7 && c >= cells-7) || (r >= cells-7 && c < 7);
        const isInner = (r >= 2 && r <= 4 && c >= 2 && c <= 4) || (r >= 2 && r <= 4 && c >= cells-5 && c <= cells-3) || (r >= cells-5 && r <= cells-3 && c >= 2 && c <= 4);
        row.push(isCorner || isInner || Math.random() > 0.55);
      }
      g.push(row);
    }
    return g;
  });
  const cell = size / cells;
  return (
    <div style={{background:"#fff",padding:10,borderRadius:12,display:"inline-block",boxShadow:"0 4px 20px rgba(0,0,0,0.15)"}}>
      <div style={{display:"grid",gridTemplateColumns:`repeat(${cells},${cell}px)`,gap:0}}>
        {grid.map((row,r)=>row.map((on,c)=>(
          <div key={`${r}-${c}`} style={{width:cell,height:cell,background:on?"#1a1040":"#fff"}}/>
        )))}
      </div>
    </div>
  );
}

                <div style={{margin:"0 auto 16px",display:"flex",justifyContent:"center"}}>
                  <QRGrid size={160}/>
                </div>
                <div style={{background:T.surface,borderRadius:11,padding:"12px 16px",marginBottom:14,display:"flex",justifyContent:"space-between",alignItems:"center",border:`1px solid ${T.border}`}}>
                  <div style={{textAlign:"left"}}>
                    <div style={{fontSize:11,color:T.sub}}>PromptPay Number</div>
                    <div style={{fontWeight:800,fontSize:18,color:T.text,letterSpacing:2}}>{PROMPTPAY_ID}</div>
                  </div>
                  <button onClick={()=>copy(PROMPTPAY_ID)} style={{padding:"7px 14px",borderRadius:9,border:`1px solid ${T.border}`,background:T.card,color:copied?"#34d399":"#a78bfa",fontSize:12,fontWeight:700,cursor:"pointer",transition:"all 0.2s"}}>{copied?"✓ Copied!":"Copy"}</button>
                </div>
                <div style={{textAlign:"left",marginBottom:14}}>
                  {["Open your banking app (KBank, SCB, Bangkok Bank...)","Tap 'QR Payment' or 'Scan'","Scan the QR code above",`Confirm amount: ฿${Math.round(plan.price*(billing==="yearly"?10:1)*35)}`,"Screenshot your receipt"].map((s,i)=>(
                    <div key={i} style={{display:"flex",gap:9,marginBottom:8,alignItems:"flex-start"}}>
                      <div style={{width:20,height:20,borderRadius:"50%",background:"linear-gradient(135deg,#a78bfa,#60a5fa)",display:"flex",alignItems:"center",justifyContent:"center",fontSize:11,fontWeight:700,color:"#fff",flexShrink:0}}>{i+1}</div>
                      <span style={{fontSize:13,color:T.mid,paddingTop:1}}>{s}</span>
                    </div>
                  ))}
                </div>
                <div style={{marginBottom:12}}>{inp("email","your@email.com","Your Email (to activate plan)")}</div>
                <button onClick={handlePay} style={{width:"100%",padding:"13px",borderRadius:12,border:"none",background:"linear-gradient(135deg,#34d399,#059669)",color:"#fff",fontWeight:700,fontSize:14,cursor:"pointer"}}>
                  ✅ I've Sent the Payment
                </button>
              </div>
            )}

            {/* Bank transfer */}
            {method==="bank"&&card(
              <div>
                <div style={{fontWeight:700,fontSize:15,color:T.text,marginBottom:14}}>🏦 Bank Transfer</div>
                {[
                  ["Bank","Bangkok Bank / Kasikorn (KBank)"],
                  ["Account Name","ClientAI Pro Co., Ltd."],
                  ["Account Number","123-4-56789-0"],
                  ["Amount",`${fmt(plan)} (${currency.code})`],
                  ["Reference","Your email address"],
                ].map(([l,v])=>(
                  <div key={l} style={{display:"flex",justifyContent:"space-between",alignItems:"center",padding:"10px 0",borderBottom:`1px solid ${T.border}`}}>
                    <span style={{fontSize:13,color:T.sub,flexShrink:0}}>{l}</span>
                    <div style={{display:"flex",alignItems:"center",gap:8}}>
                      <span style={{fontSize:13,fontWeight:700,color:T.text,textAlign:"right"}}>{v}</span>
                      <button onClick={()=>copy(v)} style={{padding:"3px 8px",borderRadius:6,border:`1px solid ${T.border}`,background:T.surface,color:"#a78bfa",fontSize:10,fontWeight:700,cursor:"pointer",flexShrink:0}}>Copy</button>
                    </div>
                  </div>
                ))}
                <div style={{marginTop:14,marginBottom:12}}>{inp("email","your@email.com","Your Email (for confirmation)")}</div>
                <button onClick={handlePay} style={{width:"100%",padding:"13px",borderRadius:12,border:"none",background:"linear-gradient(135deg,#a78bfa,#60a5fa)",color:"#fff",fontWeight:700,fontSize:14,cursor:"pointer"}}>
                  ✅ I've Made the Transfer
                </button>
              </div>
            )}

            {!method&&<div style={{textAlign:"center",padding:"20px",color:T.sub,fontSize:14}}>👆 Select a payment method above</div>}
          </div>
        )}

        {/* ===== PROCESSING ===== */}
        {screen==="processing"&&(
          <div style={{textAlign:"center",padding:"80px 20px"}}>
            <div style={{fontSize:48,marginBottom:20}}>
              {progress < 100 ? "⚡" : "🎉"}
            </div>
            <h2 style={{fontSize:22,fontWeight:800,marginBottom:8,color:T.text}}>
              {progress < 100 ? "Processing your payment..." : "Almost done!"}
            </h2>
            <p style={{color:T.sub,fontSize:14,marginBottom:32}}>
              {method==="card"?"Verifying card details...":method==="paypal"?"Connecting to PayPal...":method==="promptpay"?"Verifying PromptPay transfer...":"Confirming bank transfer..."}
            </p>
            <div style={{maxWidth:300,margin:"0 auto"}}>
              <div style={{background:T.surface,border:`1px solid ${T.border}`,borderRadius:100,height:8,overflow:"hidden",marginBottom:16}}>
                <div style={{height:"100%",width:`${progress}%`,background:"linear-gradient(90deg,#a78bfa,#60a5fa)",borderRadius:100,transition:"width 0.4s ease"}}/>
              </div>
              <div style={{fontSize:13,color:T.sub}}>{progress}% complete</div>
            </div>
            <div style={{marginTop:32,display:"flex",flexDirection:"column",gap:8,maxWidth:280,margin:"32px auto 0"}}>
              {[
                {label:"Securing connection",done:progress>=15},
                {label:"Verifying payment",done:progress>=55},
                {label:"Activating plan",done:progress>=90},
                {label:"Sending confirmation",done:progress>=100},
              ].map(s=>(
                <div key={s.label} style={{display:"flex",alignItems:"center",gap:10}}>
                  <div style={{width:18,height:18,borderRadius:"50%",background:s.done?"linear-gradient(135deg,#34d399,#059669)":T.surface,border:`1px solid ${s.done?"#34d399":T.border}`,display:"flex",alignItems:"center",justifyContent:"center",fontSize:10,color:"#fff",transition:"all 0.3s",flexShrink:0}}>{s.done?"✓":""}</div>
                  <span style={{fontSize:13,color:s.done?T.text:T.sub,transition:"color 0.3s"}}>{s.label}</span>
                </div>
              ))}
            </div>
          </div>
        )}

        {/* ===== SUCCESS ===== */}
        {screen==="success"&&(
          <div style={{textAlign:"center",padding:"40px 20px"}}>
            <div style={{fontSize:64,marginBottom:16,animation:"bounce 0.5s ease"}}>🎉</div>
            <div style={{display:"inline-flex",alignItems:"center",gap:6,background:"rgba(52,211,153,0.1)",border:"1px solid rgba(52,211,153,0.3)",borderRadius:100,padding:"5px 14px",fontSize:12,color:"#34d399",fontWeight:700,marginBottom:16}}>
              ✓ Payment Confirmed
            </div>
            <h1 style={{fontSize:26,fontWeight:800,margin:"0 0 8px",color:T.text}}>
              {plan?.price===0?"Welcome to ClientAI!":"You're all set!"}
            </h1>
            <p style={{fontSize:15,color:T.sub,marginBottom:6}}>
              {plan?.price===0?"Your free plan is active and ready to use.":
               `ClientAI ${plan?.name} is now active on your account.`}
            </p>
            {form.email&&<p style={{fontSize:13,color:T.sub,marginBottom:28}}>Confirmation sent to <strong style={{color:T.text}}>{form.email}</strong></p>}

            {card(<div>
              <div style={{fontWeight:700,fontSize:14,color:T.text,marginBottom:12}}>🚀 Your plan includes:</div>
              {plan?.features.map(f=>(
                <div key={f} style={{display:"flex",gap:9,marginBottom:8,alignItems:"center"}}>
                  <span style={{color:"#34d399",fontWeight:700,fontSize:13}}>✓</span>
                  <span style={{fontSize:14,color:T.mid}}>{f}</span>
                </div>
              ))}
            </div>,{maxWidth:400,margin:"0 auto 20px"})}

            <div style={{display:"flex",gap:10,justifyContent:"center",flexWrap:"wrap"}}>
              <button onClick={()=>{setScreen("plans");setPlan(null);setMethod(null);setForm({name:"",email:"",card:"",expiry:"",cvv:"",paypalEmail:""});setErrors({});}}
                style={{padding:"11px 22px",borderRadius:11,border:`1px solid ${T.border}`,background:T.card,color:T.text,fontWeight:600,fontSize:14,cursor:"pointer"}}>
                ← View Plans
              </button>
              <button style={{padding:"11px 22px",borderRadius:11,border:"none",background:"linear-gradient(135deg,#a78bfa,#60a5fa)",color:"#fff",fontWeight:700,fontSize:14,cursor:"pointer",boxShadow:"0 4px 20px rgba(167,139,250,0.3)"}}>
                ⚡ Open ClientAI Pro →
              </button>
            </div>

            {countdown!==null&&countdown>0&&(
              <p style={{marginTop:16,fontSize:12,color:T.sub}}>Auto-redirecting in {countdown}s...</p>
            )}
          </div>
        )}
      </div>
    </div>
  );
}
