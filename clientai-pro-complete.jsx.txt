function QRVisual({size=150}) {
  const cells=21;
  const [grid]=useState(()=>{
    const g=[];
    for(let r=0;r<cells;r++){const row=[];for(let c=0;c<cells;c++){const corner=(r<7&&c<7)||(r<7&&c>=cells-7)||(r>=cells-7&&c<7);const inner=(r>=2&&r<=4&&c>=2&&c<=4)||(r>=2&&r<=4&&c>=cells-5&&c<=cells-3)||(r>=cells-5&&r<=cells-3&&c>=2&&c<=4);row.push(corner||inner||Math.random()>0.55);}g.push(row);}return g;
  });
  const cell=size/cells;
  return <div style={{background:"#fff",padding:8,borderRadius:10,display:"inline-block"}}><div style={{display:"grid",gridTemplateColumns:`repeat(${cells},${cell}px)`}}>{grid.map((row,r)=>row.map((on,c)=><div key={`${r}-${c}`} style={{width:cell,height:cell,background:on?"#1a1040":"#fff"}}/>))}</div></div>;
}

          {showAddFu&&card(<div><div style={{fontWeight:700,fontSize:14,color:T.text,marginBottom:10}}>New Follow-up</div><div style={{display:"flex",flexDirection:"column",gap:9}}>{inp("clientName","Name","Client Name",newFu,setNewFu)}{inp("company","Company","Company",newFu,setNewFu)}{inp("contact","Phone/email","Contact",newFu,setNewFu)}{inp("lastContacted","e.g. June 20","Last Contacted",newFu,setNewFu)}{inp("followupDate","e.g. June 27","Follow-up Date",newFu,setNewFu)}{sel("reason",["Follow up on proposal","Check interest","Share portfolio","Reconnect","Other"],"Reason",newFu,setNewFu)}{inp("notes","Notes","Notes",newFu,setNewFu)}</div><div style={{display:"flex",gap:7,marginTop:11}}><button onClick={()=>setShowAddFu(false)} style={{flex:1,padding:"8px",borderRadius:8,fontSize:13,cursor:"pointer",border:`1px solid ${T.border}`,background:T.card,color:T.mid,fontWeight:600}}>Cancel</button><button onClick={()=>{if(newFu.clientName){setFollowups(f=>[{...newFu,id:Date.now()},...f]);setNewFu({clientName:"",company:"",contact:"",lastContacted:"",followupDate:"",reason:"Follow up on proposal",notes:""});setShowAddFu(false);}}} style={{flex:2,padding:"8px",borderRadius:8,fontSize:13,cursor:"pointer",border:"none",background:"linear-gradient(135deg,#a78bfa,#60a5fa)",color:"#fff",fontWeight:700}}>Save</button></div></div>)}
          {followups.length===0&&!showAddFu&&card(<div style={{textAlign:"center",padding:"24px 0"}}><div style={{fontSize:30,marginBottom:8}}>🔔</div><p style={{color:T.sub,fontSize:13,margin:0}}>No follow-ups yet.</p></div>)}
          {followups.map(fu=>card(<div key={fu.id}><div style={{display:"flex",justifyContent:"space-between",alignItems:"flex-start"}}><div><div style={{fontWeight:700,fontSize:14,color:T.text}}>{fu.clientName} · {fu.company}</div><div style={{fontSize:12,color:T.sub}}>{fu.reason}</div>{fu.followupDate&&<div style={{fontSize:12,color:"#f59e0b",fontWeight:600,marginTop:2}}>📅 {fu.followupDate}</div>}</div><button onClick={()=>setFollowups(f=>f.filter(x=>x.id!==fu.id))} style={{padding:"3px 9px",borderRadius:6,fontSize:10,cursor:"pointer",border:"1px solid rgba(220,38,38,0.3)",background:"rgba(220,38,38,0.08)",color:"#dc2626",fontWeight:600}}>Remove</button></div><button onClick={()=>generateFollowup(fu)} disabled={fuLoading&&selFu?.id===fu.id} style={{marginTop:9,width:"100%",padding:"8px",borderRadius:9,fontSize:13,cursor:"pointer",border:"none",background:"linear-gradient(135deg,#a78bfa,#60a5fa)",color:"#fff",fontWeight:700}}>{fuLoading&&selFu?.id===fu.id?"Generating...":"✍️ Generate Message"}</button>{selFu?.id===fu.id&&fuMsg&&<div style={{marginTop:9}}><pre style={{whiteSpace:"pre-wrap",fontSize:13,color:T.mid,lineHeight:1.7,margin:"0 0 9px",fontFamily:"inherit",background:T.input,borderRadius:9,padding:11,border:`1px solid ${T.inputBorder}`}}>{fuMsg}</pre><div style={{display:"flex",gap:6}}><button onClick={()=>copyText(fuMsg,"fu")} style={{flex:1,padding:"7px",borderRadius:7,fontSize:12,cursor:"pointer",border:`1px solid ${T.border}`,background:T.card,color:copied==="fu"?"#16a34a":T.mid,fontWeight:600}}>{copied==="fu"?"✓":"📋"}</button><button onClick={()=>openWhatsApp(fuMsg)} style={{flex:2,padding:"7px",borderRadius:7,fontSize:12,cursor:"pointer",border:"none",background:"linear-gradient(135deg,#25d366,#128c7e)",color:"#fff",fontWeight:700}}>Send WhatsApp →</button></div></div>}</div>))}
        </div>}

        {/* NAMING */}
        {screen==="naming"&&<div><h2 style={{fontSize:20,fontWeight:800,marginBottom:5,color:T.text}}>✨ Name AI</h2><p style={{color:T.sub,fontSize:13,marginBottom:14}}>Find the perfect business name</p>{card(<div style={{display:"flex",flexDirection:"column",gap:10}}>{inp("industry","e.g. Web Design","Industry",namingForm,setNamingForm)}{sel("style",["Modern","Classic","Playful","Luxury","Minimal","Bold","Local","International"],"Name Style",namingForm,setNamingForm)}{inp("keywords","e.g. gold, creative, fast","Keywords (optional)",namingForm,setNamingForm)}{inp("location","e.g. Bangkok, Global","Market",namingForm,setNamingForm)}<AIBtn onClick={generateNames} loading={namingLoading} disabled={!namingForm.industry} label={`✨ Generate Names in ${selectedLang.label}`}/></div>)}<ResultBox text={namingResult} id="naming"/></div>}

        {/* COMPETITOR */}
        {screen==="competitor"&&<div><h2 style={{fontSize:20,fontWeight:800,marginBottom:5,color:T.text}}>🔍 Competitor Analysis</h2><p style={{color:T.sub,fontSize:13,marginBottom:14}}>Beat your competition with AI</p>{card(<div style={{display:"flex",flexDirection:"column",gap:10}}>{inp("yourService","e.g. Website Design","Your Service",compForm,setCompForm)}{inp("yourLocation","e.g. Bangkok","Your Location",compForm,setCompForm)}{inp("competitor","e.g. Wix, local agencies","Competitor (optional)",compForm,setCompForm)}<AIBtn onClick={generateCompetitor} loading={compLoading} disabled={!compForm.yourService||!compForm.yourLocation} label={`🔍 Analyse in ${selectedLang.label}`}/></div>)}<ResultBox text={compResult} id="comp"/></div>}

      </div>

      {/* Bottom Nav */}
      <div style={{position:"fixed",bottom:0,left:0,right:0,background:T.nav,backdropFilter:"blur(20px)",borderTop:`1px solid ${T.border}`,display:"flex",padding:"7px 2px 11px",zIndex:100,overflowX:"auto"}}>
        {TOOLS.map(s=><button key={s.id} onClick={()=>setScreen(s.id)} style={{flex:"0 0 auto",minWidth:58,display:"flex",flexDirection:"column",alignItems:"center",gap:1,padding:"5px 3px",borderRadius:9,border:"none",background:screen===s.id?(dark?"rgba(167,139,250,0.15)":"rgba(124,58,237,0.1)"):"transparent",cursor:"pointer",color:screen===s.id?"#a78bfa":T.sub,fontWeight:screen===s.id?700:400,transition:"all 0.2s"}}><span style={{fontSize:16}}>{s.icon}</span><span style={{fontSize:9}}>{s.label}</span></button>)}
      </div>
    </div>
  );
}
